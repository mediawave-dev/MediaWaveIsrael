  • ריחוף ופוקוסהסימון נפרש מימין לשמאל ב-360ms, cubic-bezier(.22,.86,.24,1): יד מהירה ונחיתה זהירה. ביטול הסימון מהיר יותר, 150ms, כמו קליק במקום אחר.
  • הסמןמופיע רק בפוקוס, רק כשהסימון נוחת, תמיד אחרי המילה האחרונה ולעולם לא מעל אות. מהבהב בקצב של עורך (530ms דולק, 530ms כבוי) חמש פעמים, ואז נעצר דולק. אין לולאה במנוחה.
  • לחיצההסימון מתרחב לכל השורה ב-120ms והשורה שוקעת פיקסל (scale .985, 90ms). נגיעה קצרה מחזיקה את הלחיצה לפחות 180ms, כדי שגם אצבע תראה את האישור.
  • שולחיםShift+← על חזרת מקש: הסימון גדל בעשרה צעדים של 114ms, מחזיק 360ms, וחוזר. בעמודה מופיעה נקודת "לא נשמר" של העורך.
  • נשלחכל השורה נשארת מסומנת (300ms), ובעמודה נכתב סימן וי בשישה פריימים של ~42ms. הסמן חונה אחרי המילה האחרונה ולא מהבהב: העבודה נגמרה.
  • שגיאההסימון מתגלגל אחורה פעם אחת, כמו ביטול (210ms, חמישה צעדים). בעמודה סימן אבחון. בלי רעידה ובלי צבע: הכפתור נשאר כפתור שאפשר ללחוץ עליו שוב.

איך בונים את זה על StateButton

  • cls="sel". ה-render מחזיר שורה אחת: עמודה (מספר השורה וסימני המצב), הטקסט האמיתי, עותק בתוך הסימון, סמן, ועותק מלא של השורה בשכבת האישור. כל השכבות חוץ מהטקסט האמיתי aria-hidden.
  • אין צורך ב-onPoint: הסימון לא עוקב אחרי העכבר. כל המצבים הם CSS על data-state, data-force, data-pressed.
  • תוספת אחת ל-StateButton: לחיצה מינימלית של 180ms (setTimeout ב-onPointerUp), ו-prop בשם ln למספר השורה. מספר השורה = הסדר של ה-CTA בעמוד מלמעלה. [להחלטה: נתי]
  • מאפיינים מונפשים: clip-path על .sel__hl ו-.sel__commit, opacity על .sel__caret, transform על הכפתור. כל שכבה עם בעלים אחד.
// SelectButton.tsx
export function SelectButton({ ln = '01', ...props }: LookButtonProps & { ln?: string }) {
  return (
    <StateButton {...props} cls="sel" render={(label, state) => {
      const row = (copy: boolean) => (<>
        <span className="sel__gutter" aria-hidden="true">
          <span className="sel__ln">{ln}</span><DirtyDot/><Check/><Bang/>
        </span>
        <span className="sel__line">
          {copy ? <span className="sel__text">{label}</span> : <>
            <span className="sel__text">{label}</span>   {/* the real, announced label */}
            <span className="sel__hl" aria-hidden="true"><span className="sel__text">{label}</span></span>
            <span className="sel__caret" aria-hidden="true" />
          </>}
        </span>
      </>)
      return (<>{row(false)}<span className="sel__commit" aria-hidden="true">{row(true)}</span></>)
    }} />
  )
}
// select-button.css = the ".sel" block of this page, as is. The copies inside .sel__hl / .sel__commit
// get the label as plain text (aria-hidden), so only the real span carries aria-live.
